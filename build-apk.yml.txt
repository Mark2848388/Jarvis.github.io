name: Build APK
on:
  push:
    branches: [main]
  workflow_dispatch:
jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: '17'
      - uses: gradle/actions/setup-gradle@v4
        with:
          gradle-version: '8.9'
      - name: Prepare project
        run: |
          unzip -q -o android.zip -d android
          mkdir -p android/app/src/main/assets/www
          cp index.html app.js style.css manifest.json icon-192.png icon-512.png android/app/src/main/assets/www/
          for d in mdpi hdpi xhdpi xxhdpi xxxhdpi; do mkdir -p android/app/src/main/res/mipmap-$d; cp icon-192.png android/app/src/main/res/mipmap-$d/ic_launcher.png; done
      - name: Build
        working-directory: android
        run: gradle assembleDebug --no-daemon
      - uses: actions/upload-artifact@v4
        with:
          name: JARVIS-apk
          path: android/app/build/outputs/apk/debug/app-debug.apk
